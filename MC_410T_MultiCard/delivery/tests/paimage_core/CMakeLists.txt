cmake_minimum_required(VERSION 3.21)
project(PaimageRecoveryChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_library(paimage_source STATIC ../../src/PaimageAcquisition/SourceCore.cpp ../../src/PaimageAcquisition/OutputQueues.cpp ../../src/PaimageAcquisition/ControlState.cpp ../../src/PaimageAcquisition/FrameConverter.cpp ../../src/PaimageAcquisition/PhysicalRoundNormalizer.cpp)
target_include_directories(paimage_source PUBLIC ../../include)
target_compile_options(paimage_source PRIVATE -Wall -Wextra -Wpedantic)
add_library(paimage_discovery STATIC ../../src/PaimageAcquisition/CardDiscovery.cpp ../../src/DiagnosticRecorder.cpp)
target_include_directories(paimage_discovery PUBLIC ../../include)
target_compile_definitions(paimage_discovery PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(paimage_discovery PUBLIC Qt6::Core ws2_32 iphlpapi)
target_compile_options(paimage_discovery PRIVATE -Wall -Wextra -Wpedantic)
add_executable(paimage_discovery_checks discovery_checks.cpp)
target_link_libraries(paimage_discovery_checks PRIVATE paimage_discovery paimage_source Qt6::Core)
add_test(NAME paimage_discovery_checks COMMAND paimage_discovery_checks)
add_executable(paimage_discovery_socket_checks discovery_socket_checks.cpp)
target_compile_definitions(paimage_discovery_socket_checks PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(paimage_discovery_socket_checks PRIVATE paimage_discovery paimage_source Qt6::Core ws2_32)
add_test(NAME paimage_discovery_socket_checks COMMAND paimage_discovery_socket_checks)
set_tests_properties(paimage_discovery_checks paimage_discovery_socket_checks PROPERTIES RUN_SERIAL TRUE)
get_target_property(_qt_qmake_location Qt6::qmake IMPORTED_LOCATION)
get_filename_component(_qt_bin_dir "${_qt_qmake_location}" DIRECTORY)
get_filename_component(_mingw_bin_dir "${CMAKE_CXX_COMPILER}" DIRECTORY)
foreach(_discovery_test paimage_discovery_checks paimage_discovery_socket_checks)
    add_custom_command(TARGET ${_discovery_test} POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:Qt6::Core>"
            "${_mingw_bin_dir}/libgcc_s_seh-1.dll" "${_mingw_bin_dir}/libstdc++-6.dll"
            "${_mingw_bin_dir}/libwinpthread-1.dll"
            "$<TARGET_FILE_DIR:${_discovery_test}>"
        COMMAND_EXPAND_LISTS VERBATIM)
endforeach()
add_executable(paimage_core_checks core_checks.cpp)
target_link_libraries(paimage_core_checks PRIVATE paimage_source)
add_test(NAME paimage_core_checks COMMAND paimage_core_checks)
add_executable(paimage_output_checks output_checks.cpp)
target_link_libraries(paimage_output_checks PRIVATE paimage_source)
add_test(NAME paimage_output_checks COMMAND paimage_output_checks)
add_executable(physical_round_normalizer_test physical_round_normalizer_test.cpp)
target_link_libraries(physical_round_normalizer_test PRIVATE paimage_source)
add_test(NAME physical_round_normalizer_test COMMAND physical_round_normalizer_test)
add_executable(paimage_control_checks control_checks.cpp)
target_link_libraries(paimage_control_checks PRIVATE paimage_source)
add_test(NAME paimage_control_checks COMMAND paimage_control_checks)
add_library(paimage_socket STATIC ../../src/PaimageAcquisition/SocketReceiver.cpp ../../src/PaimageAcquisition/SocketTimestamp.cpp ../../src/PaimageAcquisition/TraceWriter.cpp ../../src/PaimageAcquisition/TimingWriter.cpp ../../src/PaimageAcquisition/LoopLog.cpp ../../src/PaimageAcquisition/OutputWorkers.cpp ../../src/PaimageAcquisition/ControlSocket.cpp)
target_compile_definitions(paimage_socket PUBLIC PAIMAGE_SOCKET_TEST_SEAM)
target_link_libraries(paimage_socket PUBLIC paimage_source ws2_32)
target_compile_options(paimage_socket PRIVATE -Wall -Wextra -Wpedantic)
add_executable(paimage_socket_replay socket_replay.cpp)
target_link_libraries(paimage_socket_replay PRIVATE paimage_socket)
add_test(NAME paimage_socket_short COMMAND paimage_socket_replay --seconds 2 --samples 5000 --out replay-short)
set_tests_properties(paimage_socket_short PROPERTIES RUN_SERIAL TRUE)
add_executable(paimage_trace_checks trace_checks.cpp)
target_link_libraries(paimage_trace_checks PRIVATE paimage_socket)
add_test(NAME paimage_trace_checks COMMAND paimage_trace_checks)
add_executable(paimage_worker_checks worker_checks.cpp)
target_link_libraries(paimage_worker_checks PRIVATE paimage_socket)
add_test(NAME paimage_worker_checks COMMAND paimage_worker_checks)
add_executable(paimage_protocol_checks protocol_checks.cpp)
target_link_libraries(paimage_protocol_checks PRIVATE paimage_socket)
add_test(NAME paimage_protocol_checks COMMAND paimage_protocol_checks)
set_tests_properties(paimage_protocol_checks PROPERTIES RUN_SERIAL TRUE)
add_executable(paimage_conversion_checks conversion_checks.cpp)
target_link_libraries(paimage_conversion_checks PRIVATE paimage_source)
add_test(NAME paimage_conversion_checks COMMAND paimage_conversion_checks)
add_executable(paimage_looplog_checks looplog_checks.cpp)
target_link_libraries(paimage_looplog_checks PRIVATE paimage_socket)
add_test(NAME paimage_looplog_checks COMMAND paimage_looplog_checks)
